#!/bin/bash

# Renk kodları
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
RED='\033[0;31m'
NC='\033[0m' # No Color

# Kaynak, hedef klasörleri ve dosya uzantısını al
SOURCE_DIR="$1"
TARGET_DIR="$2"
EXTENSION="$3"

# Parametreler eksikse hata ver
if [ -z "$SOURCE_DIR" ] || [ -z "$TARGET_DIR" ] || [ -z "$EXTENSION" ]; then
  echo -e "${RED}Hata: Parametreler eksik.${NC}"
  echo -e "Kullanım: ${YELLOW}./donustur_hepsi_bir_arada.sh <kaynak_klasor> <hedef_klasor> <dosya_uzantisi>${NC}"
  echo -e "Örnek: ${YELLOW}./donustur_hepsi_bir_arada.sh ./eski_diagramlar ./yeni_diagramlar .txt${NC}"
  exit 1
fi

# Hedef klasör yoksa oluştur
mkdir -p "$TARGET_DIR"

echo "Dönüşüm işlemi başlatıldı..."
echo "--------------------------------------------------"

# Kaynak klasördeki belirtilen uzantılı tüm dosyaları bul ve dönüştür
find "$SOURCE_DIR" -type f -name "*${EXTENSION}" | while read -r file; do
  
  # Dosya adını ve yeni yolu belirle
  filename=$(basename "$file")
  new_filename="${filename%.*}.puml"
  output_path="$TARGET_DIR/$new_filename"
  
  echo -e "${GREEN}Dönüştürülüyor:${NC} $file -> $output_path"
  
  # Yeni dosyayı oluştur ve @startuml etiketini ekle
  echo "@startuml" > "$output_path"
  echo "" >> "$output_path"
  
  # Dosya içeriğini cat ile oku ve sed komut zincirinden geçirerek dönüştür
  # -E bayrağı, modern (extended) regular expression kullanımını sağlar.
  cat "$file" | \
  sed -E 's/^Title:(.*)/title\1/' | \
  sed -E 's/^[Nn]ote\b/note/' | \
  sed -E 's/^participant[[:space:]]+([^[:space:]]+)[[:space:]]+as[[:space:]]+(.*)/participant "\2" as \1/' \
  >> "$output_path"
  
  # Dosyanın sonuna @enduml etiketini ekle
  echo "" >> "$output_path"
  echo "@enduml" >> "$output_path"
  
done

echo "--------------------------------------------------"
echo -e "${GREEN}İşlem tamamlandı!${NC}"
