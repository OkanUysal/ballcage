#!/bin/bash

# Renk kodları
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
RED='\033[0;31m'
NC='\033[0m' # No Color

# Kullanıcıdan URL ön ekini (prefix) al
URL_PREFIX="$1"

# Eğer prefix verilmemişse, kullanım talimatını göster ve çık
if [ -z "$URL_PREFIX" ]; then
  echo -e "${RED}Hata: URL ön eki (prefix) belirtilmedi.${NC}"
  echo -e "Kullanım: ${YELLOW}./readme_detayli.sh \"https://sunucu.com/reponuz/raw/main/\"${NC}"
  exit 1
fi

# Prefix'in sonunda / olduğundan emin ol. Yoksa ekle.
if [[ "$URL_PREFIX" != */ ]]; then
  URL_PREFIX="${URL_PREFIX}/"
fi

echo "URL ön eki olarak ayarlandı: $URL_PREFIX"
echo "--------------------------------------------------"

# Mevcut dizinden itibaren tüm alt klasörleri bul
find . -mindepth 1 -type d -not -path '*/\.*' | while read -r dir; do
  
  # O klasörde en az bir PNG dosyası var mı diye kontrol et
  if ls -1 "$dir"/*.png >/dev/null 2>&1; then
    
    # --- YENİ KISIM BAŞLANGICI ---
    # Açıklamaları tutmak için bir associative array (sözlük) tanımla
    declare -A aciklamalar
    ACIKLAMA_DOSYASI="$dir/aciklamalar.txt"
    
    # Eğer aciklamalar.txt dosyası varsa, oku ve sözlüğe yükle
    if [ -f "$ACIKLAMA_DOSYASI" ]; then
      echo -e "${YELLOW}Açıklama dosyası bulundu:${NC} $ACIKLAMA_DOSYASI"
      while IFS=':' read -r anahtar deger; do
        # Anahtar ve değerin başındaki/sonundaki boşlukları temizle
        anahtar_temiz=$(echo "$anahtar" | xargs)
        deger_temiz=$(echo "$deger" | xargs)
        if [ -n "$anahtar_temiz" ]; then
          aciklamalar["$anahtar_temiz"]="$deger_temiz"
        fi
      done < "$ACIKLAMA_DOSYASI"
    fi
    # --- YENİ KISIM SONU ---
    
    README_FILE="$dir/README.md"
    echo -e "${GREEN}README.md oluşturuluyor:${NC} $README_FILE"
    
    DIR_NAME=$(basename "$dir")
    echo "# $DIR_NAME Görüntüleri" > "$README_FILE"
    
    # Klasör içindeki tüm PNG dosyaları için döngü başlat
    for png_file in "$dir"/*.png; do
      
      filename=$(basename "$png_file")
      relative_path=${png_file#./}
      full_url="${URL_PREFIX}${relative_path}"
      
      # --- YENİ KISIM BAŞLANGICI ---
      # Resim için açıklamayı sözlükten al
      aciklama=${aciklamalar["$filename"]}
      
      echo "" >> "$README_FILE" # Estetik için boşluk bırak
      
      # Eğer açıklama varsa, README'ye ekle
      if [ -n "$aciklama" ]; then
        echo "$aciklama" >> "$README_FILE"
        echo "" >> "$README_FILE" # Açıklama ile resim arasına boşluk koy
      fi
      # --- YENİ KISIM SONU ---
      
      # Markdown formatında resim satırını ekle
      echo "![${filename}](${full_url})" >> "$README_FILE"
      
    done
  fi
done

echo "--------------------------------------------------"
echo -e "${GREEN}İşlem tamamlandı!${NC}"
